private var unsettledSelection:
    Parameter.Mode? = null

internal fun selectionRealizationFinished(
    parameter: Parameter.Mode,
) {
    synchronized(selectionObservationLock) {
        if (
            unsettledSelection === parameter
        ) {
            unsettledSelection = null
        }
    }
}

// observation boundary

    unsettled != null &&
    observed != unsettled.mode
) {
    return@synchronized currentState
}
